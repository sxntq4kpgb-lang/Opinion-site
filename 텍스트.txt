            try {
                await signInAnonymously(auth);
            } catch (error) {
                console.error("Firebase Auth Error:", error);
                showToast("인증 서버에 연결할 수 없습니다. 콘솔을 확인해 주세요.", "error");
            }
        };

        // Auth State Observer
        onAuthStateChanged(auth, (user) => {
            currentUser = user;
            if (user) {
                setupRealtimeIssues();
                handleURLQuery();
            }
        });

        // Trigger Auth
        initAuthentication();

        // Setup real-time listener for "issues" collection
        function setupRealtimeIssues() {
            if (!currentUser) return;
            
            if (unsubscribeList) unsubscribeList();

            const issuesCollection = collection(db, 'issues');
            
            unsubscribeList = onSnapshot(issuesCollection, (snapshot) => {
                const loadedIssues = [];
                snapshot.forEach((doc) => {
                    loadedIssues.push({ id: doc.id, ...doc.data() });
                });

                // If empty DB, populate dummy database entries for initial engagement
                if (loadedIssues.length === 0) {
                    populateDefaultMockupData();
                    return;
                }

                issues = loadedIssues;
                renderIssues();

                // If actively viewing detail, hot-reload the UI components
                if (activeIssueId) {
                    const activeIssue = issues.find(i => i.id === activeIssueId);
                    if (activeIssue) {
                        renderVoteSection(activeIssue);
                        renderComments(activeIssue);
                    }
                }
            }, (error) => {
                console.error("Firestore sync error:", error);
                showToast("데이터를 실시간으로 가져오는 중 에러가 발생했습니다.", "error");
            });
        }

            const issuesCollection = collection(db, 'issues');

        // URL Deep Linking handler
        function handleURLQuery() {
            const urlParams = new URLSearchParams(window.location.search);
            const sharedIssueId = urlParams.get('id');
            if (sharedIssueId) {
                const checkInterval = setInterval(() => {
                    if (issues.length > 0) {
                        clearInterval(checkInterval);
                        const exists = issues.some(i => i.id === sharedIssueId);
                        if (exists) {
                            openIssueDetail(sharedIssueId);
                        } else {
                            showToast("찾으시는 의견 공유 링크가 존재하지 않습니다.", "warning");
                            history.replaceState({}, document.title, window.location.pathname);
                        }
                    }
                }, 100);
            }
        }

        // Navigation Cleanups

        // Render Issue List
        function renderIssues() {
            const grid = document.getElementById('issues-grid');
            const emptyState = document.getElementById('empty-state');
            grid.innerHTML = '';

            const filtered = issues.filter(issue => {
                const matchesCategory = currentFilter === 'all' || issue.category === currentFilter;
                const matchesSearch = issue.title.toLowerCase().includes(searchQuery) || 
                                      issue.description.toLowerCase().includes(searchQuery);
                return matchesCategory && matchesSearch;
            });

            // Order by date (Descending)
            filtered.sort((a, b) => new Date(b.date) - new Date(a.date));

            if (filtered.length === 0) {
                grid.classList.add('hidden');
                emptyState.classList.remove('hidden');
                return;
            }

            grid.classList.remove('hidden');
            emptyState.classList.add('hidden');

            filtered.forEach(issue => {
                const totalVotes = issue.agreeCount + issue.disagreeCount;
                const agreeRate = totalVotes > 0 ? Math.round((issue.agreeCount / totalVotes) * 100) : 50;
                
                const card = document.createElement('div');
                card.className = "bg-white p-6 rounded-2xl border border-slate-200 hover:border-blue-300 transition duration-300 shadow-sm hover:shadow-md cursor-pointer flex flex-col justify-between";
                card.onclick = () => openIssueDetail(issue.id);

                card.innerHTML = `
                    <div class="space-y-3">
                        <div class="flex items-center justify-between">
                            <span class="bg-slate-100 text-slate-600 text-xs px-2.5 py-1 rounded-lg font-semibold">${issue.category}</span>
                            <span class="text-xs text-slate-400">${issue.date}</span>
                        </div>
                        <h3 class="font-bold text-slate-900 text-lg hover:text-blue-600 transition line-clamp-1">${issue.title}</h3>
                        <p class="text-slate-500 text-sm line-clamp-3 leading-relaxed">${issue.description}</p>
                    </div>

                    <div class="border-t border-slate-100 pt-4 mt-6">
                        <div class="flex items-center justify-between text-xs text-slate-500 mb-2">
                            <span><i class="fa-solid fa-users mr-1"></i> 투표자: <strong>${totalVotes}</strong>명</span>
                            <span>찬성 비율: <strong class="text-emerald-600">${agreeRate}%</strong></span>
                        </div>
                        
                        <div class="w-full bg-slate-100 h-2 rounded-full overflow-hidden flex mb-4">
                            <div class="bg-emerald-500 h-full" style="width: ${agreeRate}%"></div>
                            <div class="bg-rose-500 h-full" style="width: ${100 - agreeRate}%"></div>
                        </div>

                        <div class="flex items-center justify-between text-xs text-slate-400 font-semibold">
                            <span class="flex items-center space-x-1"><i class="fa-regular fa-comment-dots text-blue-500 text-sm"></i> <span>댓글 ${(issue.comments || []).length}</span></span>
                            <span class="text-blue-600 flex items-center space-x-1"><span>토론 참여하기</span> <i class="fa-solid fa-chevron-right text-[10px]"></i></span>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        // Open Issue Detail
        window.openIssueDetail = function(id) {
            const issue = issues.find(i => i.id === id);
            if (!issue) return;

            activeIssueId = id;
            
            const newUrl = `${window.location.pathname}?id=${id}`;
            history.replaceState({ id: id }, document.title, newUrl);

            showSection('detail');

            document.getElementById('detail-category').textContent = issue.category;
            document.getElementById('detail-date').innerHTML = `<i class="fa-regular fa-clock mr-1"></i> 작성일: ${issue.date}`;
            document.getElementById('detail-title').textContent = issue.title;
            document.getElementById('detail-description').textContent = issue.description;

            renderVoteSection(issue);
            renderComments(issue);
        };

        // Render Vote progress bar
        function renderVoteSection(issue) {
            const total = (issue.agreeCount || 0) + (issue.disagreeCount || 0);
            const agreeRate = total > 0 ? Math.round(((issue.agreeCount || 0) / total) * 100) : 50;
            const disagreeRate = total > 0 ? 100 - agreeRate : 50;

            document.getElementById('detail-agree-count').textContent = `${issue.agreeCount || 0}표`;
            document.getElementById('detail-disagree-count').textContent = `${issue.disagreeCount || 0}표`;
            document.getElementById('detail-total-votes').textContent = total;

            document.getElementById('percent-agree').textContent = `${agreeRate}%`;
            document.getElementById('percent-disagree').textContent = `${disagreeRate}%`;

            document.getElementById('bar-agree').style.width = `${agreeRate}%`;
            document.getElementById('bar-disagree').style.width = `${disagreeRate}%`;
        }

        // Cast Vote to DB
        window.castVote = async function(isAgree) {
            if (!currentUser) return;

            const issue = issues.find(i => i.id === activeIssueId);
            if (!issue) return;

            const votedKey = `voted_${activeIssueId}`;
            if (sessionStorage.getItem(votedKey)) {
                showToast("이미 이 제안에 소중한 권리를 행사하셨습니다.", "warning");
                return;
            }

            const issueDocRef = doc(db, 'issues', activeIssueId);
            
            try {
                if (isAgree) {
                    await updateDoc(issueDocRef, {
                        agreeCount: (issue.agreeCount || 0) + 1
                    });
                } else {
                    await updateDoc(issueDocRef, {
                        disagreeCount: (issue.disagreeCount || 0) + 1
                    });
                }
                sessionStorage.setItem(votedKey, 'true');
                showToast("투표가 실시간 반영되었습니다!", "success");
            } catch (error) {
                console.error("Voting Update Error:", error);
                showToast("투표 저장 중 네트워크 오류가 발생했습니다.", "error");
            }
        };

        // Render Comments list
        function renderComments(issue) {
            const commentsContainer = document.getElementById('comments-container');
            const countBadge = document.getElementById('comment-count-badge');
            
            const comments = issue.comments || [];
            commentsContainer.innerHTML = '';
            countBadge.textContent = comments.length;

            if (comments.length === 0) {
                commentsContainer.innerHTML = `
                    <div class="h-full flex flex-col items-center justify-center text-slate-300 py-12">
                        <i class="fa-regular fa-comments text-4xl mb-2"></i>
                        <span class="text-sm">등록된 주민의 한마디가 없습니다.</span>
                        <span class="text-xs">첫 토론의 의견을 제안해 주세요!</span>
                    </div>
                `;
                return;
            }

            comments.forEach(comment => {
                let badgeHTML = '';
                if (comment.stance === 'agree') {
                    badgeHTML = `<span class="bg-emerald-50 text-emerald-600 text-[10px] px-2 py-0.5 rounded-full font-bold">찬성</span>`;
                } else if (comment.stance === 'disagree') {
                    badgeHTML = `<span class="bg-rose-50 text-rose-600 text-[10px] px-2 py-0.5 rounded-full font-bold">반대</span>`;
                }

                const card = document.createElement('div');
                card.className = "bg-slate-50 border border-slate-200/60 p-4 rounded-xl space-y-2 relative transition hover:bg-slate-100/60";
                card.innerHTML = `
                    <div class="flex items-center justify-between">
                        <div class="flex items-center space-x-2">
                            <span class="font-bold text-slate-700 text-sm">${comment.author}</span>
                            ${badgeHTML}
                        </div>
                        <span class="text-[10px] text-slate-400">${comment.date}</span>
                    </div>
                    <p class="text-slate-600 text-sm leading-relaxed">${comment.text}</p>
                    <button onclick="deleteComment('${comment.id}')" class="absolute top-2 right-3 text-slate-300 hover:text-rose-500 transition-colors text-xs p-1" title="삭제">
                        <i class="fa-solid fa-trash-can"></i>
                    </button>
                `;
                commentsContainer.appendChild(card);
            });

            commentsContainer.scrollTop = commentsContainer.scrollHeight;
        }

        // Post Comment
        window.handleCommentSubmit = async function(event) {
            event.preventDefault();
            if (!currentUser) return;

            const authorInput = document.getElementById('comment-author');
            const contentInput = document.getElementById('comment-content');
            const stanceInput = document.querySelector('input[name="comment-stance"]:checked');

            const issue = issues.find(i => i.id === activeIssueId);
            if (!issue) return;

            const newComment = {
                id: `c-${Date.now()}-${Math.random().toString(36).substr(2, 4)}`,
                author: authorInput.value.trim(),
                text: contentInput.value.trim(),
                stance: stanceInput.value,
                date: new Date().toISOString().split('T')[0]
            };

            const updatedComments = [...(issue.comments || []), newComment];
            const issueDocRef = doc(db, 'issues', activeIssueId);

            try {
                await updateDoc(issueDocRef, {
                    comments: updatedComments
                });

                contentInput.value = '';
                authorInput.value = '';
                document.querySelector('input[name="comment-stance"][value="neutral"]').checked = true;
                showToast("소중한 의견이 동기화되었습니다.", "success");
            } catch (error) {
                console.error("Add Comment Error:", error);
                showToast("네트워크 장애로 의견을 등록할 수 없습니다.", "error");
            }
        };

        // Remove Comment
        window.deleteComment = async function(commentId) {
            if (!currentUser) return;

            const issue = issues.find(i => i.id === activeIssueId);
            if (!issue) return;

            const filteredComments = (issue.comments || []).filter(c => c.id !== commentId);
            const issueDocRef = doc(db, 'issues', activeIssueId);

            try {
                await updateDoc(issueDocRef, {
                    comments: filteredComments
                });
                showToast("한마디를 삭제 처리하였습니다.", "warning");
            } catch (error) {
                console.error("Delete Comment Error:", error);
                showToast("한마디 삭제에 실패했습니다.", "error");
            }
        };

        // Create New Issue Card
        window.handleCreateIssue = async function(event) {
            event.preventDefault();
            if (!currentUser) return;

            const category = document.getElementById('form-category').value;
            const title = document.getElementById('form-title').value.trim();
            const description = document.getElementById('form-description').value.trim();

            } catch (error) {
                console.error("Add Issue Error:", error);
                showToast("제안 등록 중 데이터베이스 오류가 발생했습니다.", "error");
            }
        };

        // Copy Shareable link to Clipboard
        window.copyShareLink = function() {
            if (!activeIssueId) return;
            const shareUrl = window.location.href;
            
            const tempInput = document.createElement('input');
            tempInput.value = shareUrl;
            document.body.appendChild(tempInput);
            tempInput.select();

                const successful = document.execCommand('copy');

                    showToast("클립보드 접근 권한이 부족합니다.", "warning");

            } catch (err) {
                showToast("링크 복사 과정 중 에러가 발생했습니다.", "error");
            }
            document.body.removeChild(tempInput);
